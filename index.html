// App State let currentState = 'language'; // language, onboarding, main, detail let currentContext = null; let selectedPlaceId = null; // DOM Elements const appDiv = document.getElementById('app'); // Render Functions function render() { appDiv.innerHTML = ''; if (currentState === 'language') { appDiv.innerHTML = renderLanguageScreen(); } else if (currentState === 'onboarding') { appDiv.innerHTML = renderOnboardingScreen(); } else if (currentState === 'main') { appDiv.innerHTML = renderMainScreen(); } else if (currentState === 'detail') { appDiv.innerHTML = renderDetailScreen(); } } // Handlers window.goToOnboarding = () => { currentState = 'onboarding'; render(); }; window.goToMain = (context) => { currentContext = context; currentState = 'main'; render(); }; window.goToDetail = (id) => { selectedPlaceId = id; currentState = 'detail'; render(); }; window.goBack = () => { if (currentState === 'detail') { currentState = 'main'; } else if (currentState === 'main') { currentState = 'onboarding'; } render(); }; window.setFilter = (context) => { currentContext = context; render(); }; // Screen Components function renderLanguageScreen() { return `
